// ─── Asset resolution (robust: normalises names, retries lazily) ─────────────

function normalise(name) {
  return name
    .replace(/^\.\//, '')
    .replace(/^assets\//, '')
    .replace(/[\s_-]+/g, '_')
    .toLowerCase();
}

function findUploaded(list, filename) {
  if (!list || !list.length) return null;
  const norm = normalise(filename);
  for (const item of list) {
    if (normalise(item.name) === norm) return item;
  }
  return null;
}

// Resolver: returns dataUrl from uploaded assets if found, otherwise fallbackPath
function resolveModelUrl(filename, fallbackPath) {
  const found = findUploaded(window.UPLOADED_3D_MODELS, filename);
  if (!found) {
    console.warn(`[resolveModelUrl] "${filename}" not found in UPLOADED_3D_MODELS — falling back to "${fallbackPath}"`);
  }
  return found ? found.dataUrl : fallbackPath;
}

function resolveAssetUrl(filename, fallbackPath) {
  const found = findUploaded(window.UPLOADED_IMAGES, filename);
  if (!found) {
    console.warn(`[resolveAssetUrl] "${filename}" not found in UPLOADED_IMAGES — falling back to "${fallbackPath}"`);
  }
  return found ? found.dataUrl : fallbackPath;
}

function makeCanvas(w, h, fn) {
  const c = document.createElement('canvas');
  c.width = w; c.height = h;
  fn(c.getContext('2d'), w, h);
  const t = new THREE.CanvasTexture(c);

// Shared helper — draws the base paper, fibre noise, cream patches, edge shadow,
// ruled lines, and an optional spine-crease AO gradient into a 512×512 context.
// spineEdge: 'left' darkens the left column (right page inner edge),
//            'right' darkens the right column (left page inner edge — but we keep
//             convention: left page spine is on its RIGHT side).
function drawBookPageBase(ctx, w, h, spineEdge) {
  // Base paper
  ctx.fillStyle = '#ede8d8';
  ctx.fillRect(0, 0, w, h);
  // Fibre noise
  for (let i = 0; i < 1200; i++) {
    const r = 128 + Math.random() * 30;
    const g = 112 + Math.random() * 26;
    const b = 82 + Math.random() * 20;
    ctx.strokeStyle = `rgba(${r},${g},${b},${0.025 + Math.random() * 0.045})`;
    ctx.lineWidth = Math.random() * 0.6;
    ctx.beginPath();
    ctx.moveTo(Math.random() * w, Math.random() * h);
    ctx.lineTo(Math.random() * w, Math.random() * h);
    ctx.stroke();
  }
  // Cream patches
  for (let i = 0; i < 10; i++) {
    const px = Math.random() * w, py = Math.random() * h;
    const rr = Math.random() * 48 + 20;
    const g = ctx.createRadialGradient(px, py, 0, px, py, rr);
    g.addColorStop(0, 'rgba(255,248,224,0.07)');
    g.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.fillStyle = g;
    ctx.fillRect(0, 0, w, h);
  }
  // Top / bottom edge shadow
  const es = ctx.createLinearGradient(0, 0, 0, h);
  es.addColorStop(0,    'rgba(70,52,24,0.08)');
  es.addColorStop(0.06, 'rgba(0,0,0,0)');
  es.addColorStop(0.94, 'rgba(0,0,0,0)');
  es.addColorStop(1,    'rgba(70,52,24,0.10)');
  ctx.fillStyle = es;
  ctx.fillRect(0, 0, w, h);
  // Ruled lines — spaced 26 px apart, starting at y=40
  const lineSpacing = 26;
  const startY = 40;
  ctx.strokeStyle = 'rgba(88,112,175,0.13)';
  ctx.lineWidth = 0.8;
  for (let y = startY; y < h - 20; y += lineSpacing) {
    ctx.beginPath();
    ctx.moveTo(12, y);
    ctx.lineTo(w - 12, y);
    ctx.stroke();
  }
  // Faint left margin rule
  ctx.strokeStyle = 'rgba(180,100,90,0.10)';
  ctx.lineWidth = 0.9;
  ctx.beginPath();
  ctx.moveTo(52, startY);
  ctx.lineTo(52, h - 20);
  ctx.stroke();

  // ── Spine-crease AO gradient ──────────────────────────────────────────────
  // A narrow darkening strip on the inner edge of each page simulates the
  // shadow that falls into the physical fold where the two pages meet.
  // 'right' spine edge  →  left page (its inner / spine side is on the RIGHT)
  // 'left'  spine edge  →  right page (its inner side is on the LEFT)
  if (spineEdge === 'right') {
    // Inner edge is at x = w (right side of texture)
    // Gradient runs from x=w (darkest) → x = w-72 (transparent)
    const aoR = ctx.createLinearGradient(w, 0, w - 72, 0);
    aoR.addColorStop(0,    'rgba(28,18,6,0.26)');   // deepest at fold
    aoR.addColorStop(0.18, 'rgba(28,18,6,0.14)');
    aoR.addColorStop(0.55, 'rgba(28,18,6,0.04)');
    aoR.addColorStop(1,    'rgba(0,0,0,0)');
    ctx.fillStyle = aoR;
    ctx.fillRect(w - 72, 0, 72, h);
    // Single-pixel darkest crease line at the very edge
    ctx.fillStyle = 'rgba(20,12,4,0.18)';
    ctx.fillRect(w - 1, 0, 1, h);
  } else if (spineEdge === 'left') {
    // Inner edge is at x = 0 (left side of texture)
    const aoL = ctx.createLinearGradient(0, 0, 72, 0);
    aoL.addColorStop(0,    'rgba(28,18,6,0.26)');
    aoL.addColorStop(0.18, 'rgba(28,18,6,0.14)');
    aoL.addColorStop(0.55, 'rgba(28,18,6,0.04)');
    aoL.addColorStop(1,    'rgba(0,0,0,0)');
    ctx.fillStyle = aoL;
    ctx.fillRect(0, 0, 72, h);
    // Crease line
    ctx.fillStyle = 'rgba(20,12,4,0.18)';
    ctx.fillRect(0, 0, 1, h);
  }
}

function makeInsetImagePageTexture(imageUrl, spineEdge, options = {}) {
  const {
    cropTop = 0,
    cropBottom = 0,
    maxWidth = 0.70,
    maxHeight = 0.74,
    offsetY = 0,
  } = options;

  const canvas = document.createElement('canvas');
  canvas.width = 1024;
  canvas.height = 1024;
  const ctx = canvas.getContext('2d');

  const renderBase = () => {
    const w = canvas.width;
    const h = canvas.height;
    ctx.clearRect(0, 0, w, h);
    drawBookPageBase(ctx, w, h, spineEdge);

    // Light wash so the imported page reads like a placed print on paper.
    const wash = ctx.createLinearGradient(0, 0, 0, h);
    wash.addColorStop(0, 'rgba(249,247,242,0.88)');
    wash.addColorStop(0.55, 'rgba(245,242,236,0.82)');
    wash.addColorStop(1, 'rgba(238,234,225,0.88)');
    ctx.fillStyle = wash;
    ctx.fillRect(18, 14, w - 36, h - 28);
  };

  renderBase();

  const tex = new THREE.CanvasTexture(canvas);
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.wrapS = THREE.ClampToEdgeWrapping;
  tex.wrapT = THREE.ClampToEdgeWrapping;
  tex.anisotropy = renderer.capabilities.getMaxAnisotropy();

  const img = new Image();
  img.crossOrigin = 'anonymous';
  img.onload = () => {
    renderBase();

    const srcX = 0;
    const srcY = Math.floor(img.naturalHeight * cropTop);
    const srcW = img.naturalWidth;
    const srcH = Math.max(1, Math.floor(img.naturalHeight * (1 - cropTop - cropBottom)));

    const boundsW = canvas.width * maxWidth;
    const boundsH = canvas.height * maxHeight;
    const scale = Math.min(boundsW / srcW, boundsH / srcH);
    const drawW = srcW * scale;
    const drawH = srcH * scale;
    const dx = (canvas.width - drawW) * 0.5;
    const dy = (canvas.height - drawH) * 0.5 + canvas.height * offsetY;

    ctx.drawImage(img, srcX, srcY, srcW, srcH, dx, dy, drawW, drawH);
    tex.needsUpdate = true;
  };
  img.src = imageUrl;

  return tex;
}

function createGuitarNoiseBuffer(ctx) {
  const length = Math.max(1, Math.floor(ctx.sampleRate * 0.08));
  const buffer = ctx.createBuffer(1, length, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < length; i++) {
    data[i] = (Math.random() * 2 - 1) * (1 - i / length);
  }
  return buffer;
}

function ensureGuitarAudioRig() {
  const AudioCtx = window.AudioContext || window.webkitAudioContext;
  if (!AudioCtx) return null;

  if (!guitarAudioRig) {
    const ctx = new AudioCtx();
    const master = ctx.createGain();
    master.gain.value = 5.0;

    const compressor = ctx.createDynamicsCompressor();
    compressor.threshold.value = -26;
    compressor.knee.value = 18;
    compressor.ratio.value = 2.5;
    compressor.attack.value = 0.003;
    compressor.release.value = 0.22;

    master.connect(compressor);
    compressor.connect(ctx.destination);

    guitarAudioRig = {
      ctx,
      master,
      noiseBuffer: createGuitarNoiseBuffer(ctx),
    };
  }

  if (guitarAudioRig.ctx.state === 'suspended') {
    guitarAudioRig.ctx.resume().catch(() => {});
  }

  return guitarAudioRig;
}

  setupRadioMedia();

    radioMedia.currentTime = 0;

  radioMedia.muted = false;
  radioMedia.volume = 1;
  const playPromise = radioMedia.play();
  if (playPromise?.catch) {
    playPromise.catch((err) => {
      console.warn('Radio play blocked:', err.message);
    });
  }
}

}

function playGuitarPluck(freq, options = {}) {
  if (!freq) return;
  const rig = ensureGuitarAudioRig();
  if (!rig) return;

  const {
    delay = 0,
    velocity = 0.58,
    brightness = 1.0,
    decay = 1.6,
  } = options;

  const now = rig.ctx.currentTime + delay + 0.001;
  const filter = rig.ctx.createBiquadFilter();
  filter.type = 'lowpass';
  filter.frequency.setValueAtTime(Math.min(5200, 1400 * brightness + freq * 9), now);
  filter.Q.value = 0.8;

  const body = rig.ctx.createBiquadFilter();
  body.type = 'peaking';
  body.frequency.setValueAtTime(Math.max(140, freq * 2.2), now);
  body.Q.value = 0.9;
  body.gain.value = 3;

  const env = rig.ctx.createGain();
  env.gain.setValueAtTime(0.0001, now);
  env.gain.linearRampToValueAtTime(0.34 * velocity, now + 0.006);
  env.gain.exponentialRampToValueAtTime(0.0001, now + decay);

  const oscA = rig.ctx.createOscillator();
  oscA.type = 'triangle';
  oscA.frequency.setValueAtTime(freq, now);
  oscA.detune.setValueAtTime((Math.random() - 0.5) * 6, now);

  const oscB = rig.ctx.createOscillator();
  oscB.type = 'sine';
  oscB.frequency.setValueAtTime(freq * 2, now);
  oscB.detune.setValueAtTime((Math.random() - 0.5) * 10, now);

  const oscBGain = rig.ctx.createGain();
  oscBGain.gain.setValueAtTime(0.16 * velocity, now);
  oscBGain.gain.exponentialRampToValueAtTime(0.0001, now + Math.min(0.42, decay * 0.4));

  const noise = rig.ctx.createBufferSource();
  noise.buffer = rig.noiseBuffer;
  const noiseFilter = rig.ctx.createBiquadFilter();
  noiseFilter.type = 'bandpass';
  noiseFilter.frequency.setValueAtTime(Math.min(5600, freq * 6 + 1200 * brightness), now);
  noiseFilter.Q.value = 0.65;

  const noiseGain = rig.ctx.createGain();
  noiseGain.gain.setValueAtTime(0.03 * velocity, now);
  noiseGain.gain.exponentialRampToValueAtTime(0.0001, now + 0.045);

  oscA.connect(filter);
  oscB.connect(oscBGain).connect(filter);
  noise.connect(noiseFilter).connect(noiseGain).connect(filter);
  filter.connect(body).connect(env).connect(rig.master);

  oscA.start(now);
  oscB.start(now);
  noise.start(now);

  const stopAt = now + decay + 0.12;
  oscA.stop(stopAt);
  oscB.stop(stopAt);
  noise.stop(now + 0.06);
}

function playGuitarStrum(options = {}) {
  const {
    direction = 'down',
    accent = 0.62,
    spread = 0.026,
  } = options;

  const chord = getGuitarCurrentChord();
  const order = direction === 'up' ? [5, 4, 3, 2, 1, 0] : [0, 1, 2, 3, 4, 5];

  order.forEach((stringIndex, i) => {
    const freq = chord.freqs[stringIndex];
    if (!freq) return;
    playGuitarPluck(freq, {
      delay: i * spread,
      velocity: Math.max(0.28, accent - i * 0.05),
      brightness: 1.06 - i * 0.05,
      decay: 1.35 - i * 0.07,
    });
  });
}

function playLampSwitchClick(nextLampOn) {
  const rig = ensureGuitarAudioRig();
  if (!rig) return;

  const now = rig.ctx.currentTime + 0.001;

  const clickNoise = rig.ctx.createBufferSource();
  clickNoise.buffer = rig.noiseBuffer;

  const clickFilter = rig.ctx.createBiquadFilter();
  clickFilter.type = 'bandpass';
  clickFilter.frequency.setValueAtTime(nextLampOn ? 2300 : 1700, now);
  clickFilter.Q.value = 1.15;

  const clickGain = rig.ctx.createGain();
  clickGain.gain.setValueAtTime(0.0001, now);
  clickGain.gain.linearRampToValueAtTime(nextLampOn ? 0.07 : 0.058, now + 0.0015);
  clickGain.gain.exponentialRampToValueAtTime(0.0001, now + 0.024);

  const thunk = rig.ctx.createOscillator();
  thunk.type = 'triangle';
  thunk.frequency.setValueAtTime(nextLampOn ? 520 : 420, now);
  thunk.frequency.exponentialRampToValueAtTime(nextLampOn ? 240 : 190, now + 0.028);

  const thunkGain = rig.ctx.createGain();
  thunkGain.gain.setValueAtTime(0.0001, now);
  thunkGain.gain.linearRampToValueAtTime(nextLampOn ? 0.044 : 0.05, now + 0.002);
  thunkGain.gain.exponentialRampToValueAtTime(0.0001, now + 0.05);

  clickNoise.connect(clickFilter).connect(clickGain).connect(rig.master);
  thunk.connect(thunkGain).connect(rig.master);

  clickNoise.start(now);
  clickNoise.stop(now + 0.035);
  thunk.start(now);
  thunk.stop(now + 0.055);
}

function buildClock() {
  const CX = -3.44;   // flush against left wall (left wall at x = -3.5)
  const CY = 2.82;    // slightly higher on the wall so it breathes above the desk line
  const CZ = 1.0;     // centre of the left wall, clearly visible from home camera

  const clockGroup = new THREE.Group();
  clockGroup.name = 'wallClock';
  clockGroup.position.set(CX, CY, CZ);
  clockGroup.rotation.y = Math.PI / 2; // face toward room (right)
  scene.add(clockGroup);

  const R = 0.26; // clock radius — slightly larger, readable from home camera

  // ── Dial face ──
  const faceMat = new THREE.MeshLambertMaterial({ color: 0xddd6ca });
  const face = new THREE.Mesh(new THREE.CircleGeometry(R, 48), faceMat);
  face.name = 'clock_face';
  face.position.set(0, 0, 0.008);
  clockGroup.add(face);

  // ── Outer rim (silver metal) — very slim torus, minimal tube thickness ──
  const rimMat = new THREE.MeshLambertMaterial({ color: 0xb3b8bc });
  const rim = new THREE.Mesh(new THREE.TorusGeometry(R, 0.008, 8, 52), rimMat);
  rim.name = 'clock_rim';
  rim.position.set(0, 0, 0.006);
  clockGroup.add(rim);

  // ── Slim wall-mount disc (back plate, matte dark) ──
  // Use a flat CircleGeometry so zero barrel thickness is visible from the front

    marker.position.set(
      Math.sin(angle) * mR,
      Math.cos(angle) * mR,
      0.009
    );
    marker.rotation.z = -angle;
    clockGroup.add(marker);
  }

  // ── Centre cap ──
  const capMat = new THREE.MeshLambertMaterial({ color: 0x111111 });
  const cap = new THREE.Mesh(new THREE.CircleGeometry(0.011, 16), capMat);
  cap.name = 'clock_cap';
  cap.position.set(0, 0, 0.012);
  clockGroup.add(cap);

  // ── Hands — pivoted at centre, tip points up at angle=0 ──
  // Hour hand
  const hourHandMat = new THREE.MeshLambertMaterial({ color: 0x111111 });
  const hourHandGeo = new THREE.BoxGeometry(0.008, R * 0.54, 0.004);
  // Shift geometry so pivot is at bottom of hand
  hourHandGeo.translate(0, R * 0.54 * 0.5, 0);
  const hourHand = new THREE.Mesh(hourHandGeo, hourHandMat);
  hourHand.name = 'clock_hourHand';
  hourHand.position.set(0, 0, 0.011);
  clockGroup.add(hourHand);

  // Minute hand
  const minHandMat = new THREE.MeshLambertMaterial({ color: 0x111111 });
  const minHandGeo = new THREE.BoxGeometry(0.005, R * 0.72, 0.004);
  minHandGeo.translate(0, R * 0.72 * 0.5, 0);
  const minHand = new THREE.Mesh(minHandGeo, minHandMat);
  minHand.name = 'clock_minHand';
  minHand.position.set(0, 0, 0.011);
  clockGroup.add(minHand);

  // Second hand — slim, accent red
  const secHandMat = new THREE.MeshLambertMaterial({ color: 0xc0392b });
  const secHandGeo = new THREE.BoxGeometry(0.003, R * 0.80, 0.004);
  secHandGeo.translate(0, R * 0.80 * 0.5, 0);
  const secHand = new THREE.Mesh(secHandGeo, secHandMat);
  secHand.name = 'clock_secHand';
  secHand.position.set(0, 0, 0.012);
  clockGroup.add(secHand);

  // Small counterweight tail on second hand
  const tailGeo = new THREE.BoxGeometry(0.003, R * 0.18, 0.004);
  tailGeo.translate(0, -(R * 0.18 * 0.5), 0);
  const secTail = new THREE.Mesh(tailGeo, secHandMat);
  secTail.name = 'clock_secTail';
  secTail.position.set(0, 0, 0.012);
  clockGroup.add(secTail);

  // Subtle soft shadow halo on wall — very faint, just enough to seat the clock
  const shadowMat = new THREE.MeshBasicMaterial({
    color: 0x000000, transparent: true, opacity: 0.06, depthWrite: false
  });
  const shadowDisc = new THREE.Mesh(new THREE.CircleGeometry(R + 0.055, 32), shadowMat);
  shadowDisc.name = 'clock_shadow';
  shadowDisc.position.set(0, 0, -0.013);
  clockGroup.add(shadowDisc);

  // Invisible proxy disc for raycasting (sits flush on face, slightly proud)
  const clockProxy = new THREE.Mesh(
    new THREE.CircleGeometry(R, 32),
    new THREE.MeshBasicMaterial({ visible: false })
  );
  clockProxy.name = 'wallClock';
  clockProxy.position.set(0, 0, 0.014);
  clockGroup.add(clockProxy);

  // Return hand refs so animate() can drive them
  return { hourHand, minHand, secHand, secTail, clockGroup, clockProxy };
}

function updateClockHands() {
  // Get current Argentina Time (UTC-3, no DST)
  const nowUTC = Date.now();
  const artMs  = nowUTC + ART_OFFSET_MS;
  const d      = new Date(artMs);

  const h  = d.getUTCHours() % 12;
  const m  = d.getUTCMinutes();
  const s  = d.getUTCSeconds();
  const ms = d.getUTCMilliseconds();

  // Smooth fractional seconds
  const secFrac = s + ms / 1000;
  const minFrac = m + secFrac / 60;
  const hourFrac = h + minFrac / 60;

  // rotation.z: negative because we want clockwise = negative Z rotation in Three.js
  // 12 o'clock = 0 rad, 3 o'clock = -PI/2
  clockHands.hourHand.rotation.z = -(hourFrac / 12) * Math.PI * 2;
  clockHands.minHand.rotation.z  = -(minFrac  / 60) * Math.PI * 2;
  clockHands.secHand.rotation.z  = -(secFrac  / 60) * Math.PI * 2;
  clockHands.secTail.rotation.z  = -(secFrac  / 60) * Math.PI * 2;
  // Keep clockGroup position in sync (inspector target)
  clockHands.clockGroup.position.set(-3.44, 2.82, 1.0);
}

const hoverMat = new THREE.MeshBasicMaterial({ color: 0xffd580, wireframe: false, transparent: true, opacity: 0.08 });
let hoverOutline = null;

function setHoverOutline(mesh) {
  clearHoverOutline();
  if (!mesh) return;
  const geo = mesh.geometry.clone();
  hoverOutline = new THREE.Mesh(geo, hoverMat.clone());
  hoverOutline.name = 'hoverOutline';
  hoverOutline.scale.copy(mesh.scale).multiplyScalar(1.04);
  mesh.add(hoverOutline);
}

function clearHoverOutline() {
  if (hoverOutline) {
    if (hoverOutline.parent) hoverOutline.parent.remove(hoverOutline);
    hoverOutline.geometry.dispose();
    hoverOutline = null;
  }
}

// keepsakePaperTexRevealed — folded paper with baked crease, ribbon stripe, and aged marks
const keepsakePaperTexRevealed = makeCanvas(256, 192, (ctx, w, h) => {
  // Base — warm aged paper
  ctx.fillStyle = '#d4c8a8';
  ctx.fillRect(0, 0, w, h);
  // Fibre noise
  for (let i = 0; i < 700; i++) {
    const r = 120 + Math.random() * 30;
    const g = 108 + Math.random() * 26;
    const b = 76 + Math.random() * 20;
    ctx.strokeStyle = `rgba(${r},${g},${b},${0.03 + Math.random() * 0.05})`;
    ctx.lineWidth = Math.random() * 0.6;
    ctx.beginPath();
    ctx.moveTo(Math.random() * w, Math.random() * h);
    ctx.lineTo(Math.random() * w, Math.random() * h);
    ctx.stroke();
  }
  // Cream tone patches
  for (let i = 0; i < 7; i++) {
    const px = Math.random() * w, py = Math.random() * h;
    const rr = Math.random() * 38 + 14;
    const g = ctx.createRadialGradient(px, py, 0, px, py, rr);
    g.addColorStop(0, 'rgba(255,248,210,0.07)');
    g.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.fillStyle = g;
    ctx.fillRect(0, 0, w, h);
  }
  // Edge shadow — suggests sheet thickness
  const es = ctx.createLinearGradient(0, 0, 0, h);
  es.addColorStop(0,    'rgba(70,50,20,0.10)');
  es.addColorStop(0.07, 'rgba(0,0,0,0)');
  es.addColorStop(0.93, 'rgba(0,0,0,0)');
  es.addColorStop(1,    'rgba(70,50,20,0.12)');
  ctx.fillStyle = es;
  ctx.fillRect(0, 0, w, h);
  const esl = ctx.createLinearGradient(0, 0, w, 0);
  esl.addColorStop(0,    'rgba(60,40,14,0.09)');
  esl.addColorStop(0.05, 'rgba(0,0,0,0)');
  esl.addColorStop(0.95, 'rgba(0,0,0,0)');
  esl.addColorStop(1,    'rgba(60,40,14,0.09)');
  ctx.fillStyle = esl;
  ctx.fillRect(0, 0, w, h);

  // Fold crease — central horizontal line with soft shadow either side
  const cy = h * 0.5;
  // Shadow above crease
  const creaseShadow = ctx.createLinearGradient(0, cy - 10, 0, cy + 10);
  creaseShadow.addColorStop(0,   'rgba(0,0,0,0)');
  creaseShadow.addColorStop(0.4, 'rgba(50,34,10,0.13)');
  creaseShadow.addColorStop(0.5, 'rgba(40,26,6,0.18)');
  creaseShadow.addColorStop(0.6, 'rgba(50,34,10,0.10)');
  creaseShadow.addColorStop(1,   'rgba(0,0,0,0)');
  ctx.fillStyle = creaseShadow;
  ctx.fillRect(0, cy - 10, w, 20);
  // Crease line itself — thin, slightly uneven
  ctx.strokeStyle = 'rgba(90,68,28,0.32)';
  ctx.lineWidth = 0.9;
  ctx.beginPath();
  ctx.moveTo(8, cy);
  ctx.bezierCurveTo(w * 0.3, cy + 0.8, w * 0.7, cy - 0.6, w - 8, cy + 0.4);
  ctx.stroke();
  // Highlight sliver just below crease — paper catches light at fold
  ctx.strokeStyle = 'rgba(255,245,215,0.18)';
  ctx.lineWidth = 0.7;
  ctx.beginPath();
  ctx.moveTo(8, cy + 1.2);
  ctx.bezierCurveTo(w * 0.3, cy + 2.0, w * 0.7, cy + 0.8, w - 8, cy + 1.6);
  ctx.stroke();

  // Second softer crease — paper folded more than once
  const cy2 = h * 0.72;
  ctx.strokeStyle = 'rgba(90,68,28,0.16)';
  ctx.lineWidth = 0.7;
  ctx.beginPath();
  ctx.moveTo(14, cy2);
  ctx.bezierCurveTo(w * 0.35, cy2 - 0.5, w * 0.65, cy2 + 0.5, w - 14, cy2 - 0.3);
  ctx.stroke();

  // Ribbon stripe — diagonal band across paper
  ctx.save();
  ctx.globalAlpha = 0.55;
  // Ribbon is a narrow diagonal band in dusty rose / muted burgundy
  const rbGrad = ctx.createLinearGradient(w * 0.36, 0, w * 0.52, 0);
  rbGrad.addColorStop(0,   'rgba(0,0,0,0)');
  rbGrad.addColorStop(0.2, 'rgba(138,64,96,0.28)');
  rbGrad.addColorStop(0.5, 'rgba(152,72,108,0.38)');
  rbGrad.addColorStop(0.8, 'rgba(138,64,96,0.28)');
  rbGrad.addColorStop(1,   'rgba(0,0,0,0)');
  ctx.fillStyle = rbGrad;
  ctx.fillRect(0, 0, w, h);
  // Thin ribbon centre-line — suggests woven texture
  ctx.strokeStyle = 'rgba(172,88,128,0.22)';
  ctx.lineWidth = 0.8;
  ctx.beginPath();
  ctx.moveTo(w * 0.44, 0); ctx.lineTo(w * 0.44, h);
  ctx.stroke();
  ctx.restore();

  // Faint aged spot — small tannin mark near one corner
  const spot = ctx.createRadialGradient(w * 0.78, h * 0.28, 0, w * 0.78, h * 0.28, 9);
  spot.addColorStop(0, 'rgba(110,80,28,0.13)');
  spot.addColorStop(1, 'rgba(0,0,0,0)');
  ctx.fillStyle = spot;
  ctx.fillRect(0, 0, w, h);
});

      const rewardModel = gltf.scene;
      const rawBounds = new THREE.Box3().setFromObject(rewardModel);
      const rawSize = rawBounds.getSize(new THREE.Vector3());
      if (rawSize.x === 0 || rawSize.y === 0 || rawSize.z === 0) return;

      const baseScale = Math.min(
        0.12 / rawSize.x,
        0.05 / rawSize.y,
        0.085 / rawSize.z
      );
      rewardModel.scale.setScalar(baseScale * 1.58);

      const scaledBounds = new THREE.Box3().setFromObject(rewardModel);
      const scaledCenter = scaledBounds.getCenter(new THREE.Vector3());
      rewardModel.position.set(-scaledCenter.x, -scaledBounds.min.y - 0.003, -scaledCenter.z - 0.01);
      rewardModel.rotation.x = -Math.PI / 2;
      rewardModel.rotation.y = -0.12;
      rewardModel.rotation.z = 0.05;

      rewardModel.traverse((child) => {
        if (!child.isMesh) return;
        child.castShadow = true;
        child.receiveShadow = true;
        const materials = Array.isArray(child.material) ? child.material : [child.material];
        materials.forEach((material) => {
          if (!material) return;
          if ('roughness' in material) material.roughness = Math.max(material.roughness ?? 0, 0.62);
          if ('metalness' in material) material.metalness = Math.min(material.metalness ?? 0, 0.08);
        });
      });

      rewardRoot.add(rewardModel);
    },
    undefined,
    (error) => {
      console.error('Failed to load keepsake reward model:', error);
    }
  );
}

function triggerKeepsakeDrawers() {
  keepsakeDrawerState.rewardActive = false;
  keepsakeDrawerState.topTarget = 0;
  keepsakeDrawerState.stuckActive = true;
  keepsakeDrawerState.stuckTime = 0;
  keepsakeDrawerState.bottomTarget = 0.12;
  keepsakePaper.material = new THREE.MeshLambertMaterial({ map: keepsakePaperTexRevealed });
}

function resetKeepsakeDrawers() {
  keepsakeDrawerState.stuckActive = false;
  keepsakeDrawerState.stuckTime = 0;
  keepsakeDrawerState.rewardActive = false;
  keepsakeDrawerState.topOffset = 0;
  keepsakeDrawerState.topTarget = 0;
  keepsakeDrawerState.bottomOffset = 0;
  keepsakeDrawerState.bottomTarget = 0;
  keepsakePaper.visible = false;
  ribbon.visible = false;
  if (keepsakeRewardObject) keepsakeRewardObject.visible = false;
  keepsakeDrawers.forEach(({ group, baseZ }) => {
    group.position.z = baseZ;
  });
}

function makeNote(x, z, rotY, w, h, layers) {
  const group = new THREE.Group();
  const sheets = [];
  const sheetShadows = [];
  // Sheet colors — very slightly varied warm whites
  const sheetColors = [0xe4dece, 0xe9e4d5, 0xedeae0, 0xf1ede4];
  const sheetOffsets = [
    { dx: 0.007, dz: -0.005, ry: 0.055 },
    { dx: -0.005, dz: 0.006, ry: -0.03 },
    { dx: 0.002, dz: 0.003, ry: 0.018 },
    { dx: 0.0, dz: 0.0, ry: 0.0 },
  ];

  for (let l = 0; l < layers; l++) {
    const off = sheetOffsets[l] || { dx: 0, dz: 0, ry: 0 };
    const notePaperMat = new THREE.MeshLambertMaterial({
      color: sheetColors[l % sheetColors.length],
      map: paperTex
    });
    // Sheets are thin — 0.004 thick
    const sheet = new THREE.Mesh(new THREE.BoxGeometry(w, 0.004, h), notePaperMat);
    sheet.position.set(off.dx, l * 0.004, off.dz);
    sheet.rotation.y = off.ry;
    sheet.receiveShadow = true;
    if (l === layers - 1) sheet.castShadow = true;
    group.add(sheet);
    sheets.push(sheet);

    // Subtle shadow under each sheet (contact shadow proxy)
    if (l > 0) {
      const shadowMat = new THREE.MeshBasicMaterial({ color: 0x1a1208, transparent: true, opacity: 0.06 });
      const shadowPlane = new THREE.Mesh(new THREE.PlaneGeometry(w + 0.01, h + 0.01), shadowMat);
      shadowPlane.rotation.x = -Math.PI / 2;
      shadowPlane.position.set(0, -0.005, 0);
      shadowPlane.renderOrder = 1;
      sheet.add(shadowPlane);
      sheetShadows.push(shadowPlane);
    }
  }

  group.position.set(x, DESK_TOP_Y + 0.055, z); // +0.05 to clear slab top surface
  group.rotation.y = rotY;
  scene.add(group);

  const topSheet = sheets[sheets.length - 1];
  const underSheet = sheets[Math.max(0, sheets.length - 2)];
  const topMarks = new THREE.Group();
  topMarks.position.y = 0.0026;
  topSheet.add(topMarks);

  // Writing lines on top sheet — varied lengths, slightly staggered
  const lineLengths = [0.78, 0.91, 0.65, 0.84, 0.72, 0.55];
  for (let i = 0; i < 6; i++) {
    const lm = new THREE.MeshBasicMaterial({
      color: 0x686050,
      transparent: true,
      opacity: 0.11 + Math.random() * 0.08
    });
    const lineW = w * lineLengths[i];
    const wLine = new THREE.Mesh(new THREE.PlaneGeometry(lineW, 0.0032), lm);
    wLine.rotation.x = -Math.PI / 2;
    // Left-justify lines with slight indent variance
    const indent = (w - lineW) * (-0.38 + Math.random() * 0.1);
    wLine.position.set(indent, 0, -h * 0.34 + i * (h * 0.148));
    topMarks.add(wLine);
  }

  return { group, topSheet, underSheet, sheetShadows };
}

function makeNotePaperBase(ctx, w, h) {
  ctx.fillStyle = '#eee6d6';
  ctx.fillRect(0, 0, w, h);

  for (let i = 0; i < 9000; i++) {
    const tone = 150 + Math.random() * 42;
    ctx.fillStyle = `rgba(${tone},${tone - 10},${tone - 20},${0.012 + Math.random() * 0.026})`;
    ctx.fillRect(Math.random() * w, Math.random() * h, 1, 1);
  }

  for (let i = 0; i < 4200; i++) {
    const px = Math.random() * w;
    const py = Math.random() * h;
    const len = Math.random() * 8 + 2;
    ctx.strokeStyle = `rgba(144,124,94,${0.018 + Math.random() * 0.03})`;
    ctx.lineWidth = Math.random() * 0.9 + 0.15;
    ctx.beginPath();
    ctx.moveTo(px, py);
    ctx.lineTo(px + (Math.random() - 0.5) * len, py + (Math.random() - 0.5) * len);
    ctx.stroke();
  }

  for (let i = 0; i < 18; i++) {
    const px = Math.random() * w;
    const py = Math.random() * h;
    const rr = 70 + Math.random() * 140;
    const glow = ctx.createRadialGradient(px, py, 0, px, py, rr);
    glow.addColorStop(0, 'rgba(255,248,232,0.042)');
    glow.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.fillStyle = glow;
    ctx.fillRect(0, 0, w, h);
  }

  const edge = ctx.createLinearGradient(0, 0, 0, h);
  edge.addColorStop(0, 'rgba(84,64,34,0.08)');
  edge.addColorStop(0.08, 'rgba(0,0,0,0)');
  edge.addColorStop(0.92, 'rgba(0,0,0,0)');
  edge.addColorStop(1, 'rgba(84,64,34,0.10)');
  ctx.fillStyle = edge;
  ctx.fillRect(0, 0, w, h);
}

function drawPencilHandLine(ctx, text, x, y, size, tilt = 0) {
  ctx.save();
  ctx.translate(x, y);
  ctx.rotate(tilt);
  ctx.textBaseline = 'alphabetic';
  ctx.lineJoin = 'round';
  ctx.lineCap = 'round';
  ctx.font = `italic ${size}px 'Bradley Hand', 'Segoe Print', 'Snell Roundhand', cursive`;

  const width = ctx.measureText(text).width;
  const boxH = size * 1.15;

  const strokePasses = [
    { alpha: 0.18, width: size * 0.034, spread: size * 0.018 },
    { alpha: 0.24, width: size * 0.025, spread: size * 0.012 },
    { alpha: 0.32, width: size * 0.016, spread: size * 0.008 },
  ];

  strokePasses.forEach((pass) => {
    for (let i = 0; i < 2; i++) {
      const dx = (Math.random() - 0.5) * pass.spread;
      const dy = (Math.random() - 0.5) * pass.spread * 0.66;
      const graphite = 72 + Math.random() * 18;
      ctx.strokeStyle = `rgba(${graphite},${graphite},${graphite + 6},${pass.alpha})`;
      ctx.lineWidth = pass.width * (0.92 + Math.random() * 0.22);
      ctx.strokeText(text, dx, dy);
    }
  });

  ctx.fillStyle = 'rgba(82,84,90,0.78)';
  ctx.fillText(text, 0, 0);

  ctx.globalCompositeOperation = 'source-atop';
  for (let i = 0; i < width * 2.8; i++) {
    const px = Math.random() * width;
    const py = -boxH * 0.82 + Math.random() * boxH;
    const graphite = 58 + Math.random() * 42;
    ctx.fillStyle = `rgba(${graphite},${graphite},${graphite + 5},${0.04 + Math.random() * 0.14})`;
    ctx.fillRect(px, py, 1 + Math.random() * 2, 1 + Math.random() * 2);
  }

  ctx.globalCompositeOperation = 'destination-out';
  for (let i = 0; i < width * 0.14; i++) {
    const px = Math.random() * width;
    const py = -boxH * 0.82 + Math.random() * boxH;
    ctx.fillStyle = `rgba(0,0,0,${0.02 + Math.random() * 0.06})`;
    ctx.fillRect(px, py, 1, 1 + Math.random());
  }

  ctx.globalCompositeOperation = 'source-over';
  ctx.restore();
}

function createPencilNoteRevealTextures() {
  const size = 2048;
  const diffuseCanvas = document.createElement('canvas');
  diffuseCanvas.width = diffuseCanvas.height = size;
  const diffuseCtx = diffuseCanvas.getContext('2d');

  const bumpCanvas = document.createElement('canvas');
  bumpCanvas.width = bumpCanvas.height = size;
  const bumpCtx = bumpCanvas.getContext('2d');

  const roughCanvas = document.createElement('canvas');
  roughCanvas.width = roughCanvas.height = size;
  const roughCtx = roughCanvas.getContext('2d');

  makeNotePaperBase(diffuseCtx, size, size);

  bumpCtx.fillStyle = '#888888';
  bumpCtx.fillRect(0, 0, size, size);
  for (let i = 0; i < 3200; i++) {
    const tone = 126 + Math.random() * 18;
    bumpCtx.fillStyle = `rgba(${tone},${tone},${tone},${0.02 + Math.random() * 0.04})`;
    bumpCtx.fillRect(Math.random() * size, Math.random() * size, 1, 1);
  }

  roughCtx.fillStyle = '#f3f3f3';
  roughCtx.fillRect(0, 0, size, size);
  for (let i = 0; i < 2800; i++) {
    const tone = 220 + Math.random() * 24;
    roughCtx.fillStyle = `rgba(${tone},${tone},${tone},${0.025 + Math.random() * 0.04})`;
    roughCtx.fillRect(Math.random() * size, Math.random() * size, 1, 1);
  }

  const lines = [
    { text: '$5000 => ?', x: 180, y: 360, size: 188, tilt: -0.022 },
    { text: '- Time to build', x: 220, y: 690, size: 160, tilt: -0.034 },
    { text: '- Trip to Bariloche', x: 180, y: 1025, size: 154, tilt: 0.016 },
    { text: '- gift for nephew', x: 205, y: 1360, size: 154, tilt: -0.02 },
    { text: '- AI Subscription', x: 185, y: 1685, size: 152, tilt: 0.018 },
  ];

  lines.forEach(({ text, x, y, size: fontSize, tilt }) => {
    drawPencilHandLine(diffuseCtx, text, x, y, fontSize, tilt);

    bumpCtx.save();
    bumpCtx.translate(x, y);
    bumpCtx.rotate(tilt);
    bumpCtx.font = `italic ${fontSize}px 'Bradley Hand', 'Segoe Print', 'Snell Roundhand', cursive`;
    bumpCtx.textBaseline = 'alphabetic';
    bumpCtx.lineJoin = 'round';
    bumpCtx.lineCap = 'round';
    bumpCtx.strokeStyle = 'rgba(94,94,94,0.85)';
    bumpCtx.lineWidth = fontSize * 0.034;
    bumpCtx.strokeText(text, 0, 0);
    bumpCtx.fillStyle = 'rgba(108,108,108,0.92)';
    bumpCtx.fillText(text, 0, 0);
    bumpCtx.restore();

    roughCtx.save();
    roughCtx.translate(x, y);
    roughCtx.rotate(tilt);
    roughCtx.font = `italic ${fontSize}px 'Bradley Hand', 'Segoe Print', 'Snell Roundhand', cursive`;
    roughCtx.textBaseline = 'alphabetic';
    roughCtx.lineJoin = 'round';
    roughCtx.lineCap = 'round';
    roughCtx.strokeStyle = 'rgba(166,166,166,0.78)';
    roughCtx.lineWidth = fontSize * 0.03;
    roughCtx.strokeText(text, 0, 0);
    roughCtx.fillStyle = 'rgba(176,176,176,0.88)';
    roughCtx.fillText(text, 0, 0);
    roughCtx.restore();
  });

  const makeTexture = (canvas, colorSpace = null) => {
    const tex = new THREE.CanvasTexture(canvas);
    tex.wrapS = tex.wrapT = THREE.ClampToEdgeWrapping;
    tex.anisotropy = renderer.capabilities.getMaxAnisotropy();
    if (colorSpace) tex.colorSpace = colorSpace;
    return tex;
  };

  return {
    map: makeTexture(diffuseCanvas, THREE.SRGBColorSpace),
    bumpMap: makeTexture(bumpCanvas),
    roughnessMap: makeTexture(roughCanvas),
  };
}

function createPillowGeometry(width, height, depth) {
  const radius = Math.min(height * 0.52, depth * 0.20, width * 0.09);
  const geo = new RoundedBoxGeometry(width, height, depth, 9, radius);
  const position = geo.attributes.position;
  const normal = geo.attributes.normal;
  const halfW = width / 2;
  const halfH = height / 2;
  const halfD = depth / 2;
  const normalVec = new THREE.Vector3();
  const targetNormal = new THREE.Vector3();

  for (let i = 0; i < position.count; i++) {
    let x = position.getX(i);
    let y = position.getY(i);
    let z = position.getZ(i);

    const nx = x / halfW;
    const ny = y / halfH;
    const nz = z / halfD;
    const topMask = THREE.MathUtils.smoothstep(ny, -0.22, 0.98);
    const bottomMask = THREE.MathUtils.smoothstep(-ny, 0.06, 1.0);
    const midBand = Math.max(0, 1 - Math.abs(ny) * 1.18);
    const rectEdge = Math.max(Math.abs(nx), Math.abs(nz) * 1.02);
    const shiftedX = nx - 0.10;
    const shiftedZ = nz + 0.04;
    const volumeCore = Math.max(0, 1 - shiftedX * shiftedX * 0.92 - shiftedZ * shiftedZ * 1.16);
    const sideMask = THREE.MathUtils.smoothstep(rectEdge, 0.34, 0.98);
    const edgeFalloff = THREE.MathUtils.smoothstep(rectEdge, 0.58, 1.03);
    const seamBand = THREE.MathUtils.smoothstep(rectEdge, 0.66, 0.80) * (1 - THREE.MathUtils.smoothstep(rectEdge, 0.86, 1.0));
    const seamInner = THREE.MathUtils.smoothstep(rectEdge, 0.54, 0.66) * (1 - THREE.MathUtils.smoothstep(rectEdge, 0.66, 0.80));
    const cornerMask = THREE.MathUtils.smoothstep(Math.sqrt(nx * nx + nz * nz * 1.05), 0.92, 1.20);
    const fullerSide = THREE.MathUtils.smoothstep(nx, -0.12, 0.92);
    const broadUneven =
      Math.sin((nx + 0.12) * Math.PI * 1.18) * 0.42 +
      Math.cos((nz - 0.04) * Math.PI * 1.62) * 0.32 +
      Math.sin((nx - nz) * Math.PI * 1.10) * 0.18;
    const dentA = Math.exp(-(((nx + 0.22) / 0.24) ** 2 + ((nz - 0.06) / 0.26) ** 2));
    const dentB = Math.exp(-(((nx - 0.10) / 0.20) ** 2 + ((nz + 0.18) / 0.20) ** 2));
    const bumpA = Math.exp(-(((nx - 0.24) / 0.24) ** 2 + ((nz - 0.04) / 0.24) ** 2));
    const bumpB = Math.exp(-(((nx + 0.04) / 0.20) ** 2 + ((nz + 0.10) / 0.22) ** 2));
    const edgeWave = Math.sin(nx * 9.0 + nz * 6.3) * Math.cos(nz * 7.8 - nx * 3.2);
    const organicRipple = Math.sin(nx * 6.4 - nz * 4.1) * Math.sin(nz * 5.2 + nx * 2.4);
    const edgeWarp = Math.sin(nx * 4.8 + nz * 8.2) * 0.5 + Math.cos(nx * 7.1 - nz * 3.9) * 0.5;
    const tiltBias = nx * 0.0030 - nz * 0.0008;

    normalVec.set(normal.getX(i), normal.getY(i), normal.getZ(i));

    const topPuff = topMask * (
      volumeCore * 0.024 +
      fullerSide * 0.0055 +
      bumpA * 0.0040 +
      bumpB * 0.0026 +
      broadUneven * 0.0030 -
      dentA * 0.0018 -
      dentB * 0.0014 -
      edgeFalloff * 0.0026 -
      cornerMask * 0.0032
    ) + tiltBias;
    const seamLift = topMask * seamBand * (0.0036 + edgeWave * 0.0010 + edgeWarp * 0.0008);
    const sideRound = sideMask * (0.010 + volumeCore * 0.019 + fullerSide * 0.005) * (0.30 + midBand * 0.70);

    x += normalVec.x * (topPuff + seamLift);
    y += normalVec.y * (topPuff + seamLift);
    z += normalVec.z * (topPuff + seamLift);

    y -= topMask * seamInner * 0.0018;
    y -= cornerMask * 0.0028;
    y += topMask * organicRipple * 0.0020 * (0.35 + volumeCore * 0.65);

    x *= 1 + sideRound * 0.85 - edgeFalloff * bottomMask * 0.018;
    z *= 1 + sideRound * 0.72 - edgeFalloff * bottomMask * 0.014;

    const bottomCompress = 0.28 * bottomMask * (0.35 + (1 - edgeFalloff) * 0.65);
    y = THREE.MathUtils.lerp(y, -halfH + 0.0045 + rectEdge * 0.0006, bottomCompress);

    y += midBand * sideMask * 0.0014;
    x += sideMask * (edgeWave * 0.0018 + organicRipple * 0.0011) * (0.26 + topMask * 0.42);
    z += sideMask * (Math.sin(nx * 8.4 - nz * 4.8) * 0.0017 + edgeWarp * 0.0009) * (0.20 + topMask * 0.48);

    position.setXYZ(i, x, y, z);

    if (ny >= 0) {
      targetNormal.set(
        nx * (0.14 + sideRound * 0.12),
        1 + volumeCore * 0.24 + seamBand * 0.10,
        nz * (0.20 + sideRound * 0.14)
      ).normalize();
      normalVec.lerp(targetNormal, 0.18 + volumeCore * 0.22 + seamBand * 0.10);
    } else {
      targetNormal.set(nx * 0.08, -1, nz * 0.10).normalize();
      normalVec.lerp(targetNormal, 0.34 + bottomCompress * 0.28);
    }

    normalVec.normalize();
    normal.setXYZ(i, normalVec.x, normalVec.y, normalVec.z);
  }

  position.needsUpdate = true;
  normal.needsUpdate = true;
  geo.computeBoundingBox();
  geo.computeBoundingSphere();
  return geo;
}

function makeMountedPhotoTexture(
  imageUrl,
  aspect,
  {
    paperTone = '#f4ede3',
    printWarmth = 0.12,
    vignetteStrength = 0.16,
    focusX = 0.5,
    focusY = 0.5,
    sideMargin = 0.06,
    topMargin = 0.055,
    bottomMargin = 0.105,
  } = {}

  const canvas = document.createElement('canvas');
  const canvasW = aspect >= 1 ? 1024 : 896;
  const canvasH = Math.round(canvasW / aspect);
  canvas.width = canvasW;
  canvas.height = canvasH;
  const ctx = canvas.getContext('2d');

  const renderBase = () => {
    ctx.clearRect(0, 0, canvasW, canvasH);
    ctx.fillStyle = paperTone;
    ctx.fillRect(0, 0, canvasW, canvasH);

    const paperShade = ctx.createLinearGradient(0, 0, canvasW, canvasH);

    ambientBlend.addColorStop(0.4, 'rgba(255,235,206,0.05)');
    ambientBlend.addColorStop(1, 'rgba(208,192,174,0.04)');
    ctx.fillStyle = ambientBlend;
    ctx.fillRect(photoX, photoY, photoW, photoH);

    const vignette = ctx.createRadialGradient(
      photoX + photoW * 0.52,
      photoY + photoH * 0.44,
      photoW * 0.18,
      photoX + photoW * 0.52,
      photoY + photoH * 0.44,
      photoW * 0.72
    );
    vignette.addColorStop(0, 'rgba(0,0,0,0)');

    ctx.fillStyle = vignette;
    ctx.fillRect(photoX, photoY, photoW, photoH);

    const edgeShadeX = ctx.createLinearGradient(photoX, 0, photoX + photoW, 0);

function makeMarkerLetterTexture(letter) {
  const canvas = document.createElement('canvas');
  canvas.width = canvas.height = 512;
  const ctx = canvas.getContext('2d');
  ctx.clearRect(0, 0, 512, 512);
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.lineJoin = 'round';
  ctx.lineCap = 'round';
  ctx.font = "italic 320px 'Marker Felt', 'Segoe Print', 'Brush Script MT', 'Trebuchet MS', sans-serif";

  const cx = 256;
  const cy = 278;

  ctx.save();
  ctx.translate(cx, cy);
  ctx.rotate(-0.08 + (letter.charCodeAt(0) % 5) * 0.018);
  ctx.translate(-cx, -cy);

  for (let i = 0; i < 3; i++) {
    ctx.shadowColor = 'rgba(184, 255, 74, 0.75)';
    ctx.shadowBlur = 28 + i * 16;
    ctx.fillStyle = `rgba(188,255,82,${0.14 + i * 0.06})`;
    ctx.fillText(letter, cx + (Math.random() - 0.5) * 5, cy + (Math.random() - 0.5) * 5);
  }

  ctx.shadowBlur = 0;
  for (let i = 0; i < 3; i++) {
    ctx.fillStyle = `rgba(198,255,96,${0.56 + Math.random() * 0.14})`;
    ctx.fillText(letter, cx + (Math.random() - 0.5) * 4, cy + (Math.random() - 0.5) * 4);
  }

  ctx.globalCompositeOperation = 'destination-out';
  for (let i = 0; i < 360; i++) {
    const px = 96 + Math.random() * 320;
    const py = 102 + Math.random() * 300;
    ctx.fillStyle = `rgba(0,0,0,${0.02 + Math.random() * 0.08})`;
    ctx.fillRect(px, py, 1 + Math.random() * 3, 1 + Math.random() * 3);
  }

  ctx.globalCompositeOperation = 'source-over';
  ctx.restore();

  const tex = new THREE.CanvasTexture(canvas);
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.wrapS = tex.wrapT = THREE.ClampToEdgeWrapping;
  tex.anisotropy = renderer.capabilities.getMaxAnisotropy();
  return tex;
}

function makeWarpedPhotoGeometry(w, h, d, bend = 0.006, cornerSkew = 0.002) {
  const geo = new THREE.BoxGeometry(w, h, d, 8, 10, 1);
  const pos = geo.attributes.position;
  for (let i = 0; i < pos.count; i++) {
    const x = pos.getX(i);
    const y = pos.getY(i);
    const z = pos.getZ(i);
    const nx = x / (w * 0.5);
    const ny = y / (h * 0.5);
    const frontMask = z > 0 ? 1 : -0.42;
    const bow = bend * (0.26 + 0.74 * (1 - Math.abs(nx))) * (0.38 + 0.62 * (1 - ny * 0.5));
    const curl = cornerSkew * nx * (0.25 + Math.abs(ny));
    pos.setXYZ(i, x, y, z + frontMask * (bow + curl));
  }
  geo.computeVertexNormals();
  return geo;
}

function createTapedPicture({
  id, x, y, z, w, h, tilt, texture, rotY = 0, rotX = 0,
  paperTint = 0xefe4d4, shadowOpacity = 0.11, bend = 0.006, cornerSkew = 0.002,
  tapeTint = 0xf0e0af, roughness = 0.985, zLift = 0,
  markerLetter = null, markerScale = 0.62, markerOffsetX = 0, markerOffsetY = 0.02, markerTilt = 0,
}) {
  const group = new THREE.Group();
  group.position.set(x, y, z + zLift);
  group.rotation.set(0.01 + rotX, rotY, tilt);
  scene.add(group);

  const shadow = new THREE.Mesh(
    new THREE.PlaneGeometry(w + 0.10, h + 0.12),
    new THREE.MeshBasicMaterial({
      color: 0x000000,
      transparent: true,
      opacity: shadowOpacity,
      alphaMap: ribbonPictureShadowTex,
      depthWrite: false,
    })
  );
  shadow.position.set(0.01, -0.012, -0.0085);
  shadow.renderOrder = 1;
  group.add(shadow);

  const contactShadow = new THREE.Mesh(
    new THREE.PlaneGeometry(w * 0.92, h * 0.22),
    new THREE.MeshBasicMaterial({
      color: 0x000000,
      transparent: true,
      opacity: shadowOpacity * 0.72,
      depthWrite: false,
    })
  );
  contactShadow.position.set(0.008, -h * 0.29, -0.0088);
  contactShadow.renderOrder = 1;
  group.add(contactShadow);

  const backer = new THREE.Mesh(
    makeWarpedPhotoGeometry(w + 0.014, h + 0.014, 0.0032, bend * 0.45, cornerSkew * 0.55),
    new THREE.MeshLambertMaterial({ color: 0xe9dfcf })
  );
  backer.position.z = -0.0066;
  backer.receiveShadow = true;
  group.add(backer);

  const photo = new THREE.Mesh(
    makeWarpedPhotoGeometry(w, h, 0.008, bend, cornerSkew),
    new THREE.MeshStandardMaterial({
      color: paperTint,
      map: texture,
      roughness,
      metalness: 0.0,
    })
  );
  photo.castShadow = true;
  photo.receiveShadow = true;
  group.add(photo);

  if (id) {
    photo.name = id;
    registerInteractable(
      photo,
      id,
      new THREE.Vector3(x + rotY * 0.24, y + 0.02, -1.78),
      new THREE.Vector3(x, y - 0.01, z - 0.01),
      { showOutline: false }
    );

    new THREE.PlaneGeometry(0.112, 0.034),

      color: 0x000000,
      transparent: true,
      opacity: 0.10,
      depthWrite: false,
    })
  );
  tapeShadow.position.set(0, h * 0.5 + 0.006, 0.0046);
  tapeShadow.renderOrder = 2;
  group.add(tapeShadow);

  const tape = new THREE.Mesh(
    new THREE.BoxGeometry(0.12, 0.042, 0.0018),
    new THREE.MeshLambertMaterial({
      color: tapeTint,
      map: tapeTex,
      transparent: true,
      opacity: 0.9,
    })
  );
  tape.position.set(0, h * 0.5 + 0.008, 0.0048);
  tape.rotation.x = -0.06;
  tape.rotation.z = -0.08 + tilt * 0.6;
  tape.castShadow = true;
  tape.receiveShadow = true;
  group.add(tape);

  if (markerLetter) {
    const markerTex = makeMarkerLetterTexture(markerLetter);
    const markerBaseX = markerOffsetX * w;
    const markerBaseY = markerOffsetY * h;

    const markerGlow = new THREE.Mesh(
      new THREE.PlaneGeometry(w * markerScale * 0.92, h * markerScale * 0.92),
      new THREE.MeshBasicMaterial({
        map: markerTex,
        transparent: true,
        opacity: 0.06,
        blending: THREE.AdditiveBlending,
        depthWrite: false,
        toneMapped: false,
      })
    );
    markerGlow.position.set(markerBaseX, markerBaseY, 0.014);
    markerGlow.rotation.z = markerTilt;
    markerGlow.visible = !lampOn;
    group.add(markerGlow);
    photoGlowMarks.push(markerGlow);

    const markerCore = new THREE.Mesh(
      new THREE.PlaneGeometry(w * markerScale * 0.76, h * markerScale * 0.76),
      new THREE.MeshBasicMaterial({
        map: markerTex,
        transparent: true,
        opacity: 0.10,
        blending: THREE.AdditiveBlending,
        depthWrite: false,
        toneMapped: false,
      })
    );
    markerCore.position.set(markerBaseX, markerBaseY, 0.0148);
    markerCore.rotation.z = markerTilt;
    markerCore.visible = !lampOn;
    group.add(markerCore);
    photoGlowMarks.push(markerCore);
  }
}

  );

  guitarRoot.add(guitarProxy);

  guitarStrumZone = new THREE.Mesh(new THREE.BoxGeometry(0.34, 0.42, 0.18), guitarHitMat);

  guitarStrumZone.position.set(0.02, 0.70, 0.12);

  guitarRoot.add(guitarFretZone);

  guitarSoundTargets.push(guitarStrumZone, guitarFretZone);

  registerInteractable(
    guitarProxy,
    'guitar',
    new THREE.Vector3(GUITAR_X + 0.72, 1.38, 0.46),
    new THREE.Vector3(GUITAR_X, 0.98, GUITAR_Z + 0.04),
    { showOutline: false }
  );

  sharedGltfLoader.load(
    guitarModelUrl,
    (gltf) => {
      const guitarModel = gltf.scene;
      const bounds = new THREE.Box3().setFromObject(guitarModel);
      const size = new THREE.Vector3();
      const center = new THREE.Vector3();
      bounds.getSize(size);
      bounds.getCenter(center);

      const targetHeight = 1.92;
      const scale = (targetHeight / Math.max(size.y, 0.001)) * 1.18;
      guitarModel.scale.setScalar(scale);

      const scaledBounds = new THREE.Box3().setFromObject(guitarModel);
      const scaledCenter = new THREE.Vector3();
      scaledBounds.getCenter(scaledCenter);

      // Keep the GLB centered and floor-grounded, but bias it slightly toward the
      // desk side instead of relying on the source pivot for contact placement.
      guitarModel.position.set(-scaledCenter.x - 0.03, -scaledBounds.min.y, -scaledCenter.z + 0.08);
      guitarModel.rotation.y = -0.02;

      guitarModel.traverse((child) => {
        if (!child.isMesh) return;
        child.castShadow = true;
        child.receiveShadow = true;

        const materials = Array.isArray(child.material) ? child.material : [child.material];
        materials.forEach((material) => {
          if (!material) return;
          if ('roughness' in material) material.roughness = 0.72;
          if ('metalness' in material) material.metalness = Math.min(material.metalness ?? 0.0, 0.04);
        });
      });

      guitarRoot.add(guitarModel);
    },
    undefined,
    (error) => {
      console.error('Failed to load acoustic guitar model:', error);
    }
  );

  const guitarFloorShadowTex = makeCanvas(512, 256, (ctx, w, h) => {
    ctx.fillStyle = '#000000';
    ctx.fillRect(0, 0, w, h);

    const paintBlob = (cx, cy, rx, ry, strength) => {
      ctx.save();
      ctx.translate(cx, cy);
      ctx.scale(rx, ry);
      const grad = ctx.createRadialGradient(0, 0, 0, 0, 0, 1);
      grad.addColorStop(0, `rgba(255,255,255,${strength})`);
      grad.addColorStop(0.55, `rgba(180,180,180,${strength * 0.45})`);
      grad.addColorStop(1, 'rgba(0,0,0,0)');
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.arc(0, 0, 1, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    };

    paintBlob(w * 0.34, h * 0.60, 126, 48, 0.92);
    paintBlob(w * 0.49, h * 0.52, 90, 34, 0.52);
    paintBlob(w * 0.68, h * 0.46, 132, 20, 0.24);
    paintBlob(w * 0.25, h * 0.66, 54, 24, 0.30);
  });

  const floorShadowMat = new THREE.MeshBasicMaterial({
    color: 0x000000,
    transparent: true,
    opacity: 0.16,
    depthWrite: false,
    alphaMap: guitarFloorShadowTex,
  });
  const floorShadow = new THREE.Mesh(new THREE.PlaneGeometry(1.04, 0.58), floorShadowMat);
  floorShadow.rotation.x = -Math.PI / 2;
  floorShadow.rotation.z = 0.18;
  floorShadow.position.set(GUITAR_X + 0.04, 0.003, GUITAR_Z + 0.12);
  scene.add(floorShadow);

  const guitarDeskShadowTex = makeCanvas(256, 512, (ctx, w, h) => {
    ctx.fillStyle = '#000000';
    ctx.fillRect(0, 0, w, h);

    const paintBlob = (cx, cy, rx, ry, strength) => {
      ctx.save();
      ctx.translate(cx, cy);
      ctx.scale(rx, ry);
      const grad = ctx.createRadialGradient(0, 0, 0, 0, 0, 1);
      grad.addColorStop(0, `rgba(255,255,255,${strength})`);
      grad.addColorStop(0.65, `rgba(170,170,170,${strength * 0.42})`);
      grad.addColorStop(1, 'rgba(0,0,0,0)');
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.arc(0, 0, 1, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    };

    paintBlob(w * 0.52, h * 0.62, 54, 118, 0.42);
    paintBlob(w * 0.42, h * 0.42, 34, 84, 0.18);
    paintBlob(w * 0.60, h * 0.78, 32, 48, 0.16);
  });

  const deskShadowMat = new THREE.MeshBasicMaterial({
    color: 0x000000,
    transparent: true,
    opacity: 0.08,
    depthWrite: false,
    alphaMap: guitarDeskShadowTex,
  });
  const deskShadow = new THREE.Mesh(new THREE.PlaneGeometry(0.34, 1.26), deskShadowMat);
  deskShadow.rotation.y = -Math.PI / 2;
  deskShadow.position.set(DESK_X + 1.57 + 0.033, 0.96, GUITAR_Z - 0.02);
  scene.add(deskShadow);
}

function getArgentinaTimeString() {
  const art = new Date(Date.now() - 3 * 60 * 60 * 1000);
  const hh = String(art.getUTCHours()).padStart(2, '0');
  const mm = String(art.getUTCMinutes()).padStart(2, '0');
  const ss = String(art.getUTCSeconds()).padStart(2, '0');
  return `${hh}:${mm}:${ss}`;
}

function isExplorationPhase() {
  return gameState.phase === 'exploring';
}

function isObjectInteractionPhase() {
  return gameState.phase === 'exploring' || gameState.phase === 'reward';
}

function getAvailableInteractables() {
  if (gameState.phase === 'reward') {
    return interactables.filter((entry) => entry.id === 'ommaReward');
  }
  if (gameState.phase === 'exploring') {
    return interactables;
  }
  return [];
}

function formatCountdown(ms) {
  const totalSeconds = Math.max(0, Math.ceil(ms / 1000));
  const minutes = String(Math.floor(totalSeconds / 60)).padStart(2, '0');
  const seconds = String(totalSeconds % 60).padStart(2, '0');
  return `${minutes}:${seconds}`;
}

function clearGameTimer() {
  if (gameState.timerHandle !== null) {
    clearInterval(gameState.timerHandle);
    gameState.timerHandle = null;
  }
}

function clearGamePhaseDelay() {
  if (gameState.phaseDelayHandle !== null) {
    clearTimeout(gameState.phaseDelayHandle);
    gameState.phaseDelayHandle = null;
  }
}

  mobileViewState.dragging = false;
  mobileViewState.pointerId = null;
  mobileViewState.suppressClickUntil = 0;
  mobileViewState.yaw = 0;
  mobileViewState.yawTarget = 0;
  mobileViewState.pitch = 0;
  mobileViewState.pitchTarget = 0;

function setGamePhase(nextPhase, options = {}) {
  gameState.phase = nextPhase;

  const introVisible = nextPhase === 'opening';
  const exploringVisible = nextPhase === 'exploring';
  const questionVisible = nextPhase === 'question';
  const resultVisible = nextPhase === 'incorrect' || nextPhase === 'correct';
  const actionVisible = nextPhase === 'incorrect' || nextPhase === 'correct';

  introOverlay.classList.toggle('visible', introVisible);
  questionOverlay.classList.toggle('visible', questionVisible);
  resultOverlay.classList.toggle('visible', resultVisible);
  roomTitle.classList.toggle('visible', exploringVisible);
  roomTimer.classList.toggle('visible', exploringVisible);
  hint.style.opacity = exploringVisible ? '1' : '0';
  retryRoundBtn.style.display = actionVisible ? 'block' : 'none';

  if (questionVisible || resultVisible || introVisible || nextPhase === 'transitioning' || nextPhase === 'restarting') {
    hideCard();
  }

  if (nextPhase === 'opening') {
    clearGameTimer();
    roomTimer.textContent = '01:00';
    answerInput.value = '';
    answerInput.disabled = false;
    answerSubmitBtn.disabled = false;
    answerResult.textContent = '';
    answerResultSubtitle.textContent = '';
    retryRoundBtn.textContent = 'Restart';
  }

  if (nextPhase === 'question') {
    clearGameTimer();
    roomTimer.textContent = '00:00';
    answerInput.disabled = false;
    answerSubmitBtn.disabled = false;
    answerResult.textContent = '';
    answerResultSubtitle.textContent = '';
    answerInput.focus();
  }

  if (nextPhase === 'incorrect' || nextPhase === 'correct') {
    clearGameTimer();
    answerInput.disabled = true;
    answerSubmitBtn.disabled = true;
    answerResult.textContent = options.resultText ?? '';
    answerResultSubtitle.textContent = options.resultSubtitle ?? '';
    retryRoundBtn.textContent = options.actionLabel ?? 'Restart';
  }
}

function updateExplorationTimer() {
  if (!isExplorationPhase()) return;
  const remaining = Math.max(0, gameState.deadlineMs - Date.now());
  roomTimer.textContent = formatCountdown(remaining);
  if (remaining <= 0) {
    setGamePhase('question');
  }
}

function beginExploration() {
  if (gameState.phase !== 'opening') return;
  gameState.deadlineMs = Date.now() + gameState.durationMs;
  setGamePhase('exploring');
  updateExplorationTimer();
  clearGameTimer();
  gameState.timerHandle = window.setInterval(updateExplorationTimer, 250);
  if (radioOn) {
    startRadioPlayback();
  }
}

function resetGameExperience() {
  clearGameTimer();
  clearGamePhaseDelay();
  stopRadioPlayback();
  radioOn = true;
  lampOn = true;
  lampLight.intensity = LAMP_KEY_INTENSITY;
  lampFill.intensity = LAMP_FILL_INTENSITY;
  syncLampDrivenMaterials();
  actionTriggered = {};
  resetIdleViewState();
  const idlePose = getIdleCameraPose();
  camState.current.copy(idlePose.position);
  camState.target.copy(idlePose.position);
  camState.lookCurrent.copy(idlePose.target);
  camState.lookTarget.copy(idlePose.target);
  camState.isZoomed = false;
  camState.activeObject = null;
  interactionState.armedObjectId = null;
  setGamePhase('opening');
}

  clearGamePhaseDelay();
  setGamePhase('restarting');
  gameState.phaseDelayHandle = window.setTimeout(() => {
    gameState.phaseDelayHandle = null;
    resetGameExperience();
  }, 280);
}

function submitGameAnswer() {
  if (gameState.phase !== 'question') return;
  const normalized = answerInput.value.replace(/[^a-z]/gi, '').toLowerCase();
  const correct = normalized === 'build';
  answerInput.disabled = true;
  answerSubmitBtn.disabled = true;

  if (correct) {
    setGamePhase('correct', {
      resultText: 'Correct.',
      resultSubtitle: 'There is one more thing.',
      actionLabel: 'Open the box',
    });
    return;
  }

  setGamePhase('transitioning');
  clearGamePhaseDelay();
  gameState.phaseDelayHandle = window.setTimeout(() => {
    gameState.phaseDelayHandle = null;
    playFailSound();
    setGamePhase('incorrect', {
      resultText: 'Not quite.',
      resultSubtitle: 'Look again.',
    });
  }, 420);
}

function moveCustomCursor(clientX, clientY) {
  if (!hasFinePointer || !cursorDot || !cursorGlow) return;
  cursorMotion.targetX = clientX;
  cursorMotion.targetY = clientY;
  if (!cursorMotion.initialized) {
    cursorMotion.initialized = true;
    cursorMotion.dotX = clientX;
    cursorMotion.dotY = clientY;
    cursorMotion.glowX = clientX;
    cursorMotion.glowY = clientY;
  }
  setCustomCursorVisible(true);
}

function updateCustomCursorMotion() {
  if (!hasFinePointer || !cursorDot || !cursorGlow || !cursorMotion.initialized) return;

  cursorMotion.dotX = THREE.MathUtils.lerp(cursorMotion.dotX, cursorMotion.targetX, 0.22);
  cursorMotion.dotY = THREE.MathUtils.lerp(cursorMotion.dotY, cursorMotion.targetY, 0.22);
  cursorMotion.glowX = THREE.MathUtils.lerp(cursorMotion.glowX, cursorMotion.targetX, 0.14);
  cursorMotion.glowY = THREE.MathUtils.lerp(cursorMotion.glowY, cursorMotion.targetY, 0.14);

  cursorDot.style.left = `${cursorMotion.dotX}px`;
  cursorDot.style.top = `${cursorMotion.dotY}px`;
  cursorGlow.style.left = `${cursorMotion.glowX}px`;
  cursorGlow.style.top = `${cursorMotion.glowY}px`;
}

function getActionButtonLabel(id) {
  const data = cardData[id];
  if (!data) return '';

  if (id === 'lamp') return lampOn ? 'Turn off' : 'Turn on';
  if (id === 'radio') return radioOn ? 'Turn off' : 'Turn on';
  if (id === 'openBook') return openBookReadMode ? 'Back to object' : 'Read closer';
  if (id === 'wallClock') return getArgentinaTimeString();
  if (id === 'keepsakeBox') return 'Try drawers';
  if (id === 'uprightBooks') return uprightBooksState.isOpen ? 'Slide back in' : 'Slide one out';
  if (id === 'handwrittenNote') return 'Slide it aside';
  if (id === 'guitar') return 'Strum';
  if (wallPhotoIds.has(id)) return data.action;
  if (id === 'poster') return 'See where it leads →';
  if (id === 'ommaReward') return 'Go to omma.build →';
  return actionTriggered[id] ? '✓ Done' : data.action;
}

function updateCardUi(id) {
  const data = cardData[id];
  if (!data) return;
  const repeatable = new Set(['lamp', 'radio', 'openBook', 'keepsakeBox', 'uprightBooks', 'handwrittenNote', 'guitar', 'poster', 'ommaReward']);

  cardTitle.textContent = data.title;
  cardMemory.textContent = data.memory;
  cardStatus.textContent = getInteractionCue(id);
  card.classList.toggle('armed', interactionState.armedObjectId === id && isObjectActionable(id));
  cardBackBtn.textContent = gameState.phase === 'reward' ? 'Start again' : '← Back to desk';

  if (id === 'wallClock') {
    stopClockDisplay();
    startClockDisplay();
  } else {
    stopClockDisplay();
    cardActionBtn.textContent = getActionButtonLabel(id);
    cardActionBtn.disabled = !isObjectActionable(id) || (!repeatable.has(id) && actionTriggered[id]);
  }
}

function hideCard() {
  card.classList.remove('visible');
  card.classList.remove('armed');
  cardStatus.textContent = '';
  if (gameState.phase === 'reward') {
    camState.activeObject = null;
    interactionState.armedObjectId = null;
    clearHoverOutline();
    hoveredMesh = null;
    setSceneCursorInteractive(false);

    return;

  // Hide any revealed details
  annotationGroup.visible = false;
}

  if (!id || !isObjectActionable(id)) return false;

  if (id === 'lamp') {
    lampOn = !lampOn;
    playLampSwitchClick(lampOn);
    lampLight.intensity = lampOn ? LAMP_KEY_INTENSITY : 0;
    lampFill.intensity = lampOn ? LAMP_FILL_INTENSITY : 0;
    syncLampDrivenMaterials();
    updateCardUi(id);
    return true;
  }
  if (id === 'radio') {
    radioOn = !radioOn;
    if (radioOn) {
      startRadioPlayback();
    } else {
      stopRadioPlayback();
    }
    updateCardUi(id);
    return true;
  }
  if (id === 'keepsakeBox') {
    triggerKeepsakeDrawers();
    updateCardUi(id);
    return true;
  }
  if (id === 'uprightBooks') {
    triggerUprightBooks();
    updateCardUi(id);
    return true;
  }
  if (id === 'handwrittenNote') {
    triggerHandwrittenNote();
    updateCardUi(id);
    return true;
  }
  if (id === 'guitar') {
    playGuitarStrum({ direction: 'down', accent: 0.68, spread: 0.024 });
    updateCardUi(id);
    return true;
  }
  if (id === 'poster') {
    const posterWindow = window.open('https://omma.build/', '_blank', 'noopener,noreferrer');
    posterWindow?.focus?.();
    updateCardUi(id);
    return true;
  }
  if (id === 'ommaReward') {
    const ommaWindow = window.open('https://omma.build/', '_blank', 'noopener,noreferrer');
    ommaWindow?.focus?.();
    updateCardUi(id);
    return true;
  }
  if (id === 'openBook') {
    setOpenBookReadMode(!openBookReadMode);
    updateCardUi(id);
    return true;
  }

  if (actionTriggered[id]) {
    updateCardUi(id);
    return false;
  }
  actionTriggered[id] = true;
  updateCardUi(id);
  return true;
}

// ─── Pointer events ───────────────────────────────────────────────────────────
let hoveredMesh = null;

renderer.domElement.addEventListener('pointerdown', (e) => {
  if (!isExplorationPhase()) return;
  if (!mobileViewState.enabled || camState.isZoomed || e.pointerType === 'mouse') return;
  mobileViewState.dragging = true;
  mobileViewState.pointerId = e.pointerId;
  mobileViewState.startX = e.clientX;
  mobileViewState.startY = e.clientY;
  mobileViewState.lastX = e.clientX;
  mobileViewState.lastY = e.clientY;
});

renderer.domElement.addEventListener('pointermove', (e) => {
  if (!isExplorationPhase()) return;
  if (!mobileViewState.enabled || !mobileViewState.dragging || mobileViewState.pointerId !== e.pointerId || camState.isZoomed) return;

  const dx = e.clientX - mobileViewState.lastX;
  const dy = e.clientY - mobileViewState.lastY;
  mobileViewState.lastX = e.clientX;
  mobileViewState.lastY = e.clientY;

  if (Math.abs(dx) + Math.abs(dy) < 0.01) return;

  mobileViewState.yawTarget = THREE.MathUtils.clamp(
    mobileViewState.yawTarget - dx * MOBILE_YAW_DRAG,
    -MOBILE_YAW_LIMIT,
    MOBILE_YAW_LIMIT
  );
  mobileViewState.pitchTarget = THREE.MathUtils.clamp(
    mobileViewState.pitchTarget - dy * MOBILE_PITCH_DRAG,
    -MOBILE_PITCH_UP_LIMIT,
    MOBILE_PITCH_DOWN_LIMIT
  );

  const dragDistance = Math.hypot(e.clientX - mobileViewState.startX, e.clientY - mobileViewState.startY);
  if (dragDistance > MOBILE_DRAG_THRESHOLD) {
    mobileViewState.suppressClickUntil = performance.now() + 260;
  }

  e.preventDefault();
});

function finishMobileSceneDrag(e) {
  if (!mobileViewState.enabled) return;
  if (e && mobileViewState.pointerId !== null && e.pointerId !== mobileViewState.pointerId) return;
  mobileViewState.dragging = false;
  mobileViewState.pointerId = null;
}

renderer.domElement.addEventListener('pointerup', finishMobileSceneDrag);
renderer.domElement.addEventListener('pointercancel', finishMobileSceneDrag);
renderer.domElement.addEventListener('pointerleave', finishMobileSceneDrag);

renderer.domElement.addEventListener('pointerdown', (e) => {
  if (!isExplorationPhase()) return;
  if (!(camState.isZoomed && camState.activeObject === 'guitar')) return;
  const hit = getGuitarSoundHitFromEvent(e);
  if (!hit) return;

  ensureGuitarAudioRig();
  guitarPointerState.active = true;
  guitarPointerState.zone = hit.object;
  guitarPointerState.lastStringIndex = null;
  guitarPointerState.lastChordIndex = null;
  guitarPointerState.moved = false;

  const local = hit.object.worldToLocal(hit.point.clone());
  if (hit.object === guitarFretZone) {
    const chordIndex = getGuitarChordIndexFromLocal(local.y);
    guitarPointerState.lastChordIndex = chordIndex;
    previewGuitarChord(chordIndex);
  } else if (hit.object === guitarStrumZone) {
    guitarPointerState.lastStringIndex = getGuitarStringIndexFromLocal(local.x);
  }

  e.preventDefault();
});

renderer.domElement.addEventListener('pointermove', (e) => {
  if (!isExplorationPhase()) return;
  if (!(camState.isZoomed && camState.activeObject === 'guitar' && guitarPointerState.active && guitarPointerState.zone)) return;

  const hit = getGuitarSoundHitFromEvent(e, [guitarPointerState.zone]);
  if (!hit) return;
  const local = hit.object.worldToLocal(hit.point.clone());

  if (hit.object === guitarStrumZone) {
    const stringIndex = getGuitarStringIndexFromLocal(local.x);
    if (guitarPointerState.lastStringIndex !== null && stringIndex !== guitarPointerState.lastStringIndex) {
      const chord = getGuitarCurrentChord();
      playGuitarPluck(chord.freqs[stringIndex], {
        velocity: 0.46,
        brightness: 1.1,
        decay: 0.95,
      });
      guitarPointerState.lastStringIndex = stringIndex;
      guitarPointerState.moved = true;
    }
  } else if (hit.object === guitarFretZone) {
    const chordIndex = getGuitarChordIndexFromLocal(local.y);
    if (chordIndex !== guitarPointerState.lastChordIndex) {
      previewGuitarChord(chordIndex);
      guitarPointerState.lastChordIndex = chordIndex;
      guitarPointerState.moved = true;
    }
  }
});

  if (!guitarPointerState.active) return;
  if (guitarPointerState.zone === guitarStrumZone && !guitarPointerState.moved) {
    playGuitarStrum({ direction: 'down', accent: 0.62, spread: 0.022 });
  }
  resetGuitarPointerState();
}

renderer.domElement.addEventListener('mousemove', (e) => {
  if (!isObjectInteractionPhase()) {
    if (hoveredMesh) {
      clearHoverOutline();
      hoveredMesh = null;
    }
    setSceneCursorInteractive(false);
    return;
  }
  if (camState.activeObject === 'guitar' && guitarPointerState.active) return;
  moveCustomCursor(e.clientX, e.clientY);
  mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
  mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
  raycaster.setFromCamera(mouse, camera);
  const activeInteractables = getAvailableInteractables();
  const hits = raycaster.intersectObjects(activeInteractables.map(i => i.mesh), false);
  if (hits.length > 0) {
    const hit = hits[0].object;
    const entry = interactables.find(i => i.mesh === hit || i.mesh.name === hit.name);
    if (hit !== hoveredMesh) {
      clearHoverOutline();
      hoveredMesh = hit;
      if (!camState.isZoomed && entry?.showOutline !== false && !isInvisibleHitTarget(hit)) {
        setHoverOutline(entry?.outlineMesh ?? hit);
      }
    }
    setSceneCursorInteractive(true);
  } else {
    if (hoveredMesh) {
      clearHoverOutline();
      hoveredMesh = null;
    }
    setSceneCursorInteractive(false);
  }
});

renderer.domElement.addEventListener('click', (e) => {
  if (!isObjectInteractionPhase()) return;
  if (mobileViewState.enabled && performance.now() < mobileViewState.suppressClickUntil) return;

  mouse.x = (e.clientX / window.innerWidth) * 2 - 1;
  mouse.y = -(e.clientY / window.innerHeight) * 2 + 1;
  raycaster.setFromCamera(mouse, camera);

  if (camState.isZoomed) {
    const activeInteractables = getAvailableInteractables();
    const hits = raycaster.intersectObjects(activeInteractables.map(i => i.mesh), false);
    const guitarSoundHits = camState.activeObject === 'guitar'
      ? raycaster.intersectObjects(guitarSoundTargets, false)
      : [];

    if (hits.length > 0) {
      const hitMesh = hits[0].object;
      const entry = interactables.find(i => i.mesh === hitMesh || i.mesh.name === hitMesh.name);
      if (entry) {
        if (entry.id !== camState.activeObject) {
          focusInteractable(entry);
        } else if (!(entry.id === 'guitar' && guitarSoundHits.length > 0)) {
          triggerObjectAction(entry.id);
        }
        return;
      }
    }

    // Clicking background while zoomed — go home
    if (guitarSoundHits.length === 0) hideCard();
    return;
  }

  const activeInteractables = getAvailableInteractables();
  const hits = raycaster.intersectObjects(activeInteractables.map(i => i.mesh), false);
  if (hits.length > 0) {
    const hitMesh = hits[0].object;
    const entry = interactables.find(i => i.mesh === hitMesh || i.mesh.name === hitMesh.name);
    if (entry) {
      focusInteractable(entry);
    }
  }
});

// ─── Animation ────────────────────────────────────────────────────────────────
let time = 0;

function animate() {
  requestAnimationFrame(animate);
  time += 0.008;

  updateCustomCursorMotion();

  if (!camState.isZoomed && mobileViewState.enabled) {
    mobileViewState.yaw = THREE.MathUtils.lerp(mobileViewState.yaw, mobileViewState.yawTarget, 0.12);
    mobileViewState.pitch = THREE.MathUtils.lerp(mobileViewState.pitch, mobileViewState.pitchTarget, 0.12);
    const idlePose = getIdleCameraPose();
    camState.target.copy(idlePose.position);
    camState.lookTarget.copy(idlePose.target);
  }

  // Smooth camera lerp
  camState.current.lerp(camState.target, 0.045);
  camState.lookCurrent.lerp(camState.lookTarget, 0.045);
  camera.position.copy(camState.current);
  camera.lookAt(camState.lookCurrent);

  // Bookshelf interaction — one book slides out while a receipt peeks up between books.
  uprightBooksState.bookOffset = THREE.MathUtils.lerp(
    uprightBooksState.bookOffset,
    uprightBooksState.bookTarget,
    0.14
  );
  uprightBooksState.receiptX = THREE.MathUtils.lerp(
    uprightBooksState.receiptX,
    uprightBooksState.receiptXTarget,
    0.16
  );
  uprightBooksState.receiptLift = THREE.MathUtils.lerp(
    uprightBooksState.receiptLift,
    uprightBooksState.receiptLiftTarget,
    0.16
  );
  uprightBooksState.receiptZ = THREE.MathUtils.lerp(
    uprightBooksState.receiptZ,
    uprightBooksState.receiptZTarget,
    0.16
  );
  if (pulledBook) {
    pulledBook.parts.forEach(({ mesh, baseZ }) => {
      mesh.position.z = baseZ + uprightBooksState.bookOffset;
    });
  }
  hiddenPaper.position.x = hiddenPaperBaseX + uprightBooksState.receiptX;
  hiddenPaper.position.y = hiddenPaperBaseY + uprightBooksState.receiptLift;
  hiddenPaper.position.z = hiddenPaperBaseZ + uprightBooksState.receiptZ;
  if (
    hiddenPaper.visible &&
    uprightBooksState.bookTarget === 0 &&
    uprightBooksState.receiptXTarget === 0 &&
    uprightBooksState.receiptLiftTarget === 0 &&
    uprightBooksState.receiptX < 0.002 &&
    uprightBooksState.receiptLift < 0.002 &&
    uprightBooksState.receiptZ < 0.002
  ) {
    hiddenPaper.visible = false;
  }

  // Handwritten note — top page slides aside to expose the more practical note underneath.
  handwrittenNoteState.x = THREE.MathUtils.lerp(handwrittenNoteState.x, handwrittenNoteState.xTarget, 0.16);
  handwrittenNoteState.y = THREE.MathUtils.lerp(handwrittenNoteState.y, handwrittenNoteState.yTarget, 0.16);
  handwrittenNoteState.z = THREE.MathUtils.lerp(handwrittenNoteState.z, handwrittenNoteState.zTarget, 0.16);
  handwrittenNoteState.rotY = THREE.MathUtils.lerp(handwrittenNoteState.rotY, handwrittenNoteState.rotYTarget, 0.16);
  note1.position.set(handwrittenNoteState.x, handwrittenNoteState.y, handwrittenNoteState.z);
  note1.rotation.y = handwrittenNoteState.rotY;

  // Keepsake drawers — top one catches and chatters, bottom one opens cleanly
  keepsakeDrawerState.bottomOffset = THREE.MathUtils.lerp(
    keepsakeDrawerState.bottomOffset,
    keepsakeDrawerState.bottomTarget,
    0.12
  );

  const topRestOffset = keepsakeDrawerState.rewardActive
    ? keepsakeDrawerState.topTarget
    : (keepsakeDrawerState.bottomTarget > 0 ? 0.008 : 0);
  if (keepsakeDrawerState.rewardActive) {
    keepsakeDrawerState.topOffset = THREE.MathUtils.lerp(
      keepsakeDrawerState.topOffset,
      keepsakeDrawerState.topTarget,
      0.12
    );
  } else if (keepsakeDrawerState.stuckActive) {
    keepsakeDrawerState.stuckTime += 0.06;
    const t = keepsakeDrawerState.stuckTime;
    const kick = Math.sin(Math.min(Math.PI * 0.95, t * 5.6)) * 0.022;
    const chatter =
      Math.sin(t * 32) * 0.005 * Math.exp(-t * 1.4) +
      Math.sin(t * 15 + 0.7) * 0.0025 * Math.exp(-t * 1.0);
    keepsakeDrawerState.topOffset = Math.max(topRestOffset, topRestOffset + kick + chatter);
    if (t > 1.0) {
      keepsakeDrawerState.stuckActive = false;
    }
  } else {
    keepsakeDrawerState.topOffset = THREE.MathUtils.lerp(
      keepsakeDrawerState.topOffset,
      topRestOffset,
      0.16
    );
  }

  keepsakeDrawers.forEach(({ group, baseZ, index }) => {
    group.position.z = baseZ + (index === 0 ? keepsakeDrawerState.topOffset : keepsakeDrawerState.bottomOffset);
  });
  keepsakePaper.visible = keepsakeDrawerState.bottomOffset > 0.06;
  if (keepsakeRewardObject) {
    keepsakeRewardObject.visible = keepsakeDrawerState.rewardActive && keepsakeDrawerState.topOffset > 0.05;
  }
  ribbon.visible = false;

  // Real-time clock update (Argentina Time, UTC-3)
  updateClockHands();

  // Very subtle lamp flicker — warmer colour shift too
  if (lampOn) {
    const flicker = Math.sin(time * 1.7) * 0.06 + Math.sin(time * 4.3) * 0.03;
    lampLight.intensity = LAMP_KEY_INTENSITY + flicker;
    lampFill.intensity = LAMP_FILL_INTENSITY + flicker * 0.4;
    bounce.intensity = DESK_BOUNCE_INTENSITY + flicker * 0.08;
  }

  renderer.render(scene, camera);
}
